Image Enhancement Techniques

CO1

Read an 8-bit image and apply: (a) Brightness improvement (b) Brightness reduction (c) Thresholding (d) Negative (e) Log transformation (f) Power Law transformation.

Image enhancement improves visual quality. Brightness shifts pixel values up/down. Thresholding converts to binary image. Negative inverts values (255–pixel). Log transform compresses dynamic range. Power-law (gamma) adjusts contrast non-linearly.

Python · OpenCV 
import cv2
import numpy as np
import matplotlib.pyplot as plt
from tkinter import Tk
from tkinter.filedialog import askopenfilename

Tk().withdraw()
file_path = askopenfilename(
    title="Select Image",
    filetypes=[("Image Files", "*.jpg *.jpeg *.png *.bmp *.gif")]
)

if file_path:
    img = cv2.imread(file_path, 0).astype(np.float32)

    # (a) Brightness Improvement
    bright = np.clip(img + 50, 0, 255).astype(np.uint8)

    # (b) Brightness Reduction
    dark = np.clip(img - 50, 0, 255).astype(np.uint8)

    # (c) Thresholding
    _, thresh = cv2.threshold(img.astype(np.uint8), 127, 255, cv2.THRESH_BINARY)

    # (d) Negative of Image
    negative = (255 - img).astype(np.uint8)

    # (e) Log Transformation
    log_img = np.uint8(255 / np.log(1 + 255) * np.log(1 + img))

    # (f) Power Law (Gamma) Transformation
    gamma = 0.5
    power_img = np.uint8(255 * (img / 255) ** gamma)

    titles = ['Original', 'Brightness+', 'Brightness-',
              'Threshold', 'Negative', 'Log', 'Power Law']
    images = [img.astype(np.uint8), bright, dark,
              thresh, negative, log_img, power_img]

    plt.figure(figsize=(16, 4))
    for i, (t, im) in enumerate(zip(titles, images)):
        plt.subplot(1, 7, i + 1)
        plt.imshow(im, cmap='gray')
        plt.title(t, fontsize=8)
        plt.axis('off')
    plt.suptitle('Practical 1 - Image Enhancement Techniques', fontsize=12, fontweight='bold')
    plt.tight_layout()
    plt.show()
else:
    print("No file selected.")

Histogram & Histogram Equalization

CO2

Read an image, plot its histogram, then perform histogram equalization and compare results.

A histogram shows the frequency of each pixel intensity (0–255). Histogram equalization redistributes pixel intensities to achieve a uniform distribution, improving global contrast especially in low-contrast or dark images.

Python · OpenCV 
import cv2

Gray Level Slicing, Bit Effect & Bit Plane Slicing

CO3

(a) Gray level slicing on cameraman image. (b) Effect of each bit on the image. (c) Extract 8 different bit planes.

Gray-level slicing highlights a specific intensity range. Bit-plane slicing decomposes an 8-bit image into 8 binary planes (bit 0 = LSB to bit 7 = MSB). Higher bit planes contain more structural/shape information; lower planes contain fine detail and noise.

Python · OpenCV 
import cv2
import numpy as np
import matplotlib.pyplot as plt
from tkinter import Tk
from tkinter.filedialog import askopenfilename

Tk().withdraw()
file_path = askopenfilename(
    title="Select 8-bit Grayscale Image",
    filetypes=[("Image Files", "*.jpg *.jpeg *.png *.bmp")]
)

if file_path:
    img = cv2.imread(file_path, 0)

    # ── (a) Gray Level Slicing ─────────────────────────────────────────
    sliced = np.where((img >= 100) & (img <= 200), 255, 0).astype(np.uint8)

    plt.figure(figsize=(8, 3))
    plt.subplot(1, 2, 1)
    plt.imshow(img, cmap='gray')
    plt.title('Original Image')
    plt.axis('off')
    plt.subplot(1, 2, 2)
    plt.imshow(sliced, cmap='gray')
    plt.title('Gray Level Sliced (100-200)')
    plt.axis('off')
    plt.suptitle('(a) Gray Level Slicing', fontweight='bold')
    plt.tight_layout()
    plt.show()

    # ── (b) Effect of Each Bit ─────────────────────────────────────────
    plt.figure(figsize=(16, 3))
    plt.suptitle('(b) Effect of Each Bit on Image', fontweight='bold')
    for bit in range(8):
        mask = np.full(img.shape, 1 << bit, dtype=np.uint8)
        bit_img = cv2.bitwise_and(img, mask)
        plt.subplot(1, 8, bit + 1)
        plt.imshow(bit_img, cmap='gray')
        plt.title(f'Bit {bit}', fontsize=8)
        plt.axis('off')
    plt.tight_layout()
    plt.show()

    # ── (c) Bit Plane Slicing ──────────────────────────────────────────
    plt.figure(figsize=(16, 3))
    plt.suptitle('(c) Bit Plane Slicing (0=LSB  to  7=MSB)', fontweight='bold')
    for bit in range(8):
        plane = ((img >> bit) & 1) * 255
        plt.subplot(1, 8, bit + 1)
        plt.imshow(plane, cmap='gray')
        plt.title(f'Plane {bit}', fontsize=8)
        plt.axis('off')
    plt.tight_layout()
    plt.show()
else:
    print("No file selected.")

Smoothing Spatial Filters

CO4

Implement various smoothing spatial filters: Mean Filter, Weighted Mean Filter, and Median Filter.

Smoothing filters reduce noise by replacing each pixel with a function of its neighbourhood. Mean filter averages all neighbours equally. Weighted mean gives higher weight to central pixels. Median filter replaces with median value — very effective for salt-and-pepper noise without blurring edges.

Python · OpenCV 
import cv2
import numpy as np
import matplotlib.pyplot as plt
from tkinter import Tk
from tkinter.filedialog import askopenfilename

Tk().withdraw()
file_path = askopenfilename(
    title="Select Image",
    filetypes=[("Image Files", "*.jpg *.jpeg *.png *.bmp")]
)

if file_path:
    img = cv2.imread(file_path, 0)

    # Add Salt & Pepper Noise for demonstration
    noisy = img.copy()
    prob = 0.02
    rnd = np.random.random(img.shape)
    noisy[rnd < prob] = 0
    noisy[rnd > 1 - prob] = 255

    # Mean Filter (3x3)
    mean_kernel = np.ones((3, 3), np.float32) / 9
    mean_img = cv2.filter2D(noisy, -1, mean_kernel)

    # Weighted Mean Filter (Gaussian-like)
    w_kernel = np.array([[1, 2, 1],
                         [2, 4, 2],
                         [1, 2, 1]], np.float32) / 16
    weighted = cv2.filter2D(noisy, -1, w_kernel)

    # Median Filter
    median_img = cv2.medianBlur(noisy, 3)

    titles = ['Original', 'Noisy Image', 'Mean Filter', 'Weighted Mean', 'Median Filter']
    images = [img, noisy, mean_img, weighted, median_img]

    plt.figure(figsize=(16, 4))
    for i, (t, im) in enumerate(zip(titles, images)):
        plt.subplot(1, 5, i + 1)
        plt.imshow(im, cmap='gray')
        plt.title(t, fontsize=9)
        plt.axis('off')
    plt.suptitle('Practical 4 - Smoothing Spatial Filters', fontsize=12, fontweight='bold')
    plt.tight_layout()
    plt.show()
else:
    print("No file selected.")

Gaussian Blur, High-Pass, Laplacian & High-Boost

CO4

(a) Gaussian 3×3 mask for blurring (b) High-pass filter mask (c) Laplacian with centre +ve and −ve (d) High-boost filtering.

Gaussian blur uses a bell-shaped kernel for smooth blurring. High-pass filter enhances edges. Laplacian (2nd derivative) detects edges; positive centre gives bright edges on dark background. High-boost = A×Original − Blurred (A > 1), which sharpens while preserving image structure.

Python · OpenCV 
import cv2
import numpy as np
import matplotlib.pyplot as plt
from tkinter import Tk
from tkinter.filedialog import askopenfilename

Tk().withdraw()
file_path = askopenfilename(
    title="Select Image",
    filetypes=[("Image Files", "*.jpg *.jpeg *.png *.bmp")]
)

if file_path:
    img = cv2.imread(file_path, 0).astype(np.float32)

    # (a) Gaussian 3x3 Blur
    gauss = cv2.GaussianBlur(img, (3, 3), 0)

    # (b) High-Pass Filter Mask
    hp_kernel = np.array([[-1, -1, -1],
                           [-1,  9, -1],
                           [-1, -1, -1]], np.float32)
    hp = cv2.filter2D(img, -1, hp_kernel)

    # (c) Laplacian - Centre Positive
    lap_pos_kernel = np.array([[0,  1, 0],
                                [1, -4, 1],
                                [0,  1, 0]], np.float32)
    lap_p = cv2.filter2D(img, -1, lap_pos_kernel)

    # (c) Laplacian - Centre Negative
    lap_neg_kernel = np.array([[ 0, -1,  0],
                                [-1,  4, -1],
                                [ 0, -1,  0]], np.float32)
    lap_n = cv2.filter2D(img, -1, lap_neg_kernel)

    # (d) High-Boost Filtering (A = 1.5)
    A = 1.5
    hb = np.clip(A * img - gauss, 0, 255)

    titles = ['Original', '(a) Gaussian Blur', '(b) High-Pass',
              '(c) Laplacian +ve', '(c) Laplacian -ve', '(d) High-Boost']
    images = [img, gauss, hp, lap_p, lap_n, hb]

    plt.figure(figsize=(16, 4))
    for i, (t, im) in enumerate(zip(titles, images)):
        plt.subplot(1, 6, i + 1)
        plt.imshow(np.clip(im, 0, 255).astype(np.uint8), cmap='gray')
        plt.title(t, fontsize=8)
        plt.axis('off')
    plt.suptitle('Practical 5 - Sharpening / Enhancement Filters', fontsize=12, fontweight='bold')
    plt.tight_layout()
    plt.show()
else:
    print("No file selected.")
Six subplots: original, Gaussian blurred, high-pass, Laplacian (+ve), Laplacian (−ve), and high-boost filtered images.

Frequency Domain Filters (LPF & HPF)

CO4

Implement various low-pass filters and high-pass filters in the frequency domain.

Frequency domain filtering uses DFT (Discrete Fourier Transform). Low-pass filters pass low frequencies (smooth/blur image). High-pass filters retain high frequencies (edges/details). Ideal filters have sharp cutoff; Gaussian filters have smooth cutoff avoiding ringing artifacts.

Python · OpenCV + NumPy 
import cv2
import numpy as np
import matplotlib.pyplot as plt
from tkinter import Tk
from tkinter.filedialog import askopenfilename

Tk().withdraw()
file_path = askopenfilename(
    title="Select Image",
    filetypes=[("Image Files", "*.jpg *.jpeg *.png *.bmp")]
)

if file_path:
    img = cv2.imread(file_path, 0)
    rows, cols = img.shape
    crow, ccol = rows // 2, cols // 2

    # Compute DFT and shift
    dft    = np.fft.fft2(img)
    dft_sh = np.fft.fftshift(dft)
    mag_sp = 20 * np.log(np.abs(dft_sh) + 1)

    D0 = 30  # Cutoff frequency

    # ── Ideal Low-Pass Filter ──────────────────────────────────────────
    mask_lp = np.zeros((rows, cols), np.uint8)
    cv2.circle(mask_lp, (ccol, crow), D0, 1, -1)
    lp_img = np.abs(np.fft.ifft2(np.fft.ifftshift(dft_sh * mask_lp))).astype(np.uint8)

    # ── Ideal High-Pass Filter ─────────────────────────────────────────
    mask_hp = 1 - mask_lp
    hp_img  = np.abs(np.fft.ifft2(np.fft.ifftshift(dft_sh * mask_hp))).astype(np.uint8)

    # ── Gaussian Low-Pass Filter ───────────────────────────────────────
    u = np.arange(cols) - ccol
    v = np.arange(rows) - crow
    U, V = np.meshgrid(u, v)
    D    = np.sqrt(U**2 + V**2)
    glp  = np.exp(-(D**2) / (2 * (D0**2)))
    glp_img = np.abs(np.fft.ifft2(np.fft.ifftshift(dft_sh * glp))).astype(np.uint8)

    # ── Gaussian High-Pass Filter ──────────────────────────────────────
    ghp     = 1 - glp
    ghp_img = np.abs(np.fft.ifft2(np.fft.ifftshift(dft_sh * ghp))).astype(np.uint8)

    titles = ['Original', 'DFT Spectrum', 'Ideal LPF', 'Ideal HPF',
              'Gaussian LPF', 'Gaussian HPF']
    images = [img, mag_sp, lp_img, hp_img, glp_img, ghp_img]

    plt.figure(figsize=(16, 4))
    for i, (t, im) in enumerate(zip(titles, images)):
        plt.subplot(1, 6, i + 1)
        plt.imshow(im, cmap='gray')
        plt.title(t, fontsize=8)
        plt.axis('off')
    plt.suptitle('Practical 6 - Frequency Domain Filters', fontsize=12, fontweight='bold')
    plt.tight_layout()
    plt.show()
else:
    print("No file selected.")
Six subplots: original, DFT spectrum, Ideal LPF, Ideal HPF, Gaussian LPF, Gaussian HPF.

Edge Detection: Sobel, Prewitt & Roberts

CO4

Implement and study the effect of different edge detection masks: Sobel, Prewitt, and Roberts.

Edge detectors compute image gradients. Sobel (3×3) is robust to noise using weighted differences. Prewitt (3×3) uses uniform weights. Roberts (2×2) is the fastest but most sensitive to noise. Gradient magnitude = √(Gx² + Gy²).

Python · OpenCV [Bug Fixed ✓] 
import cv2
import numpy as np
import matplotlib.pyplot as plt
from tkinter import Tk
from tkinter.filedialog import askopenfilename

Tk().withdraw()
file_path = askopenfilename(
    title="Select Image",
    filetypes=[("Image Files", "*.jpg *.jpeg *.png *.bmp *.gif")]
)

if file_path:
    img = cv2.imread(file_path, 0)

    # ── SOBEL ──────────────────────────────────────────────────────────
    # NOTE: Use cv2.CV_64F (capital letters) — NOT cv2.cv_64F (bug fix!)
    sobel_x = cv2.Sobel(img, cv2.CV_64F, 1, 0, ksize=3)
    sobel_y = cv2.Sobel(img, cv2.CV_64F, 0, 1, ksize=3)
    sobel   = cv2.magnitude(sobel_x, sobel_y)
    sobel   = np.uint8(np.clip(sobel, 0, 255))

    # ── PREWITT ────────────────────────────────────────────────────────
    kernal_x = np.array([[-1, 0, 1],
                          [-1, 0, 1],
                          [-1, 0, 1]], np.float32)
    kernal_y = np.array([[-1, -1, -1],
                          [ 0,  0,  0],
                          [ 1,  1,  1]], np.float32)
    prewitt_x = cv2.filter2D(img, -1, kernal_x)
    prewitt_y = cv2.filter2D(img, -1, kernal_y)
    prewitt   = cv2.add(prewitt_x, prewitt_y)

    # ── ROBERTS ────────────────────────────────────────────────────────
    kernel_x = np.array([[ 1,  0],
                          [ 0, -1]], np.float32)
    kernel_y = np.array([[ 0,  1],
                          [-1,  0]], np.float32)
    roberts_x = cv2.filter2D(img, -1, kernel_x)
    roberts_y = cv2.filter2D(img, -1, kernel_y)
    roberts   = cv2.add(roberts_x, roberts_y)

    # ── PLOT ───────────────────────────────────────────────────────────
    plt.figure(figsize=(12, 4))

    plt.subplot(1, 4, 1)
    plt.imshow(img, cmap='gray')
    plt.title('Original Image')
    plt.axis('off')

    plt.subplot(1, 4, 2)
    plt.imshow(sobel, cmap='gray')
    plt.title('Sobel')
    plt.axis('off')

    plt.subplot(1, 4, 3)
    plt.imshow(prewitt, cmap='gray')
    plt.title('Prewitt')
    plt.axis('off')

    plt.subplot(1, 4, 4)
    plt.imshow(roberts, cmap='gray')
    plt.title('Roberts')
    plt.axis('off')

    plt.suptitle('Practical 7 - Edge Detection Masks', fontsize=12, fontweight='bold')
    plt.tight_layout()
    plt.show()
else:
    print("File not found")
